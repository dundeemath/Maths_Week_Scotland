import numpy as np
import matplotlib.pyplot as plt

from shiny import App, Inputs, Outputs, Session, reactive, render, ui

# ---------------------------------------------------------------------------
# Live companion to the "Before"/"After" diagrams on the surrounding
# slides. N phases (set by the presenter with the "Oscillators" slider)
# evolve continuously by Euler-integrating the Kuramoto model
#     dtheta_i/dt = omega_i + (K/N) * sum_j sin(theta_j - theta_i)
# a fixed step at a time, driven purely by a timer -- K is the only thing
# the presenter drags live for the "watch it happen" moment. Natural
# frequencies are drawn strictly positive (all oscillators spin the same
# way, at slightly different rates) so the "Before" state reads as
# scattered, out-of-step motion rather than some dots visibly running
# backwards. The "Frequency spread" slider controls how different those
# rates are -- the wider the spread, the more coupling K it takes to pull
# everyone into phase -- which is the live bifurcation demo for the
# "Dynamical systems & bifurcation" slide later in the deck: dial the
# spread up at a fixed K and watch r collapse back down, or note how much
# higher K has to climb to re-synchronise a wider spread. Colours match
# the static SVG diagrams exactly: dashed grey unit circle (#666), teal
# oscillators (#4fd1c5), orange order-parameter arrow (#f6ad55).
# ---------------------------------------------------------------------------

N_MIN, N_MAX, N_DEFAULT = 3, 20, 8
DT = 0.01
TICK_INTERVAL = 0.15

# Natural frequencies are drawn uniformly on [MEAN - spread, MEAN + spread].
# MEAN is fixed and always positive; "spread" is the presenter-facing knob
# -- not literally a standard deviation (the draw is uniform, not normal),
# but the same role: it's the one number that sets how different the
# oscillators' natural rhythms are, from identical (spread = 0, instant,
# unconditional synchrony for any K > 0) to quite different (spread near
# its max, where low-to-moderate K visibly fails to lock the room in).
OMEGA_MEAN = 1.1
SPREAD_MIN, SPREAD_MAX, SPREAD_DEFAULT = 0.0, 1.0, 0.7  # default matches the
# fixed [0.4, 1.8] range used before this became a slider

CIRCLE_GREY = "#666666"
TEAL = "#4fd1c5"
ORANGE = "#f6ad55"
TEXT = "#e8e8e8"


def kuramoto_step(theta, omega, K, dt):
    n = len(theta)
    diff = theta[None, :] - theta[:, None]  # theta_j - theta_i
    coupling = (K / n) * np.sin(diff).sum(axis=1)
    return theta + (omega + coupling) * dt


def order_parameter(theta):
    z = np.mean(np.exp(1j * theta))
    return abs(z), np.angle(z)


def new_state(rng, n, spread):
    theta = rng.uniform(0, 2 * np.pi, n)
    lo = max(0.02, OMEGA_MEAN - spread)  # stay strictly positive even at max spread
    hi = OMEGA_MEAN + spread
    omega = rng.uniform(lo, hi, n)
    return theta, omega


app_ui = ui.page_sidebar(
    ui.sidebar(
        ui.h5("Oscillators", style=f"color:{TEXT};"),
        ui.input_slider("N", "N", min=N_MIN, max=N_MAX, value=N_DEFAULT,
                         step=1, width="100%"),
        ui.h5("Frequency spread", style=f"color:{TEXT};margin-top:1rem;"),
        ui.input_slider("spread", "spread", min=SPREAD_MIN, max=SPREAD_MAX,
                         value=SPREAD_DEFAULT, step=0.05, width="100%"),
        ui.h5("Coupling", style=f"color:{TEXT};margin-top:1rem;"),
        ui.input_slider("K", "K", min=0, max=4, value=0, step=0.05,
                         width="100%"),
        ui.input_action_button("reset", "New starting phases",
                                class_="btn-sm"),
        ui.output_ui("r_readout"),
        width=250,
        style=f"background-color:#141414;color:{TEXT};",
    ),
    ui.output_plot("circle_plot", height="420px"),
    ui.tags.style("""
        body { background-color: #191919 !important; }
        .form-range { accent-color: #f6ad55; }
    """),
)


def server(input: Inputs, output: Outputs, session: Session):
    rng = np.random.default_rng()
    theta0, omega0 = new_state(rng, N_DEFAULT, SPREAD_DEFAULT)
    theta = reactive.value(theta0)
    omega = reactive.value(omega0)

    # Re-seed whenever the oscillator count or the frequency spread
    # changes (the arrays must be resized / redrawn), or the presenter
    # asks for a fresh start. This effect only WRITES theta/omega -- it
    # never reads them back -- so it can safely depend on
    # input.N/input.spread/input.reset without the self-triggering
    # problem described below.
    @reactive.effect
    @reactive.event(input.N, input.spread, input.reset)
    def _reseed():
        th, om = new_state(rng, input.N(), input.spread())
        theta.set(th)
        omega.set(om)

    # Timer-driven animation. theta is both read and written here, which is
    # exactly the setup that caused a real self-triggering bug in the
    # fractal-fern app (see the note in analemma_py.qmd) -- the fix is the
    # same one used there: read the current state via reactive.isolate()
    # so this effect's only re-trigger source is its own invalidate_later
    # call, never its own write to theta.
    @reactive.effect
    def _tick():
        reactive.invalidate_later(TICK_INTERVAL)
        with reactive.isolate():
            th = theta()
            om = omega()
            K = input.K()
        theta.set(kuramoto_step(th, om, K, DT))

    @output
    @render.plot
    def circle_plot():
        th = theta()
        n = len(th)
        r, psi = order_parameter(th)

        fig, ax = plt.subplots(figsize=(5.4, 4.6))
        fig.patch.set_alpha(0)
        ax.set_facecolor("none")
        ax.set_xlim(-1.35, 1.35)
        ax.set_ylim(-1.35, 1.35)
        ax.set_aspect("equal")
        ax.axis("off")

        circle = plt.Circle((0, 0), 1, fill=False, edgecolor=CIRCLE_GREY,
                             linestyle="--", linewidth=1.2)
        ax.add_patch(circle)

        xs, ys = np.cos(th), np.sin(th)
        dot_size = max(35, min(150, 1200 / n))
        ax.scatter(xs, ys, s=dot_size, color=TEAL, zorder=3, edgecolor="none")

        if r < 0.04:
            ax.scatter([0], [0], s=30, color="#e8e8e8", zorder=2)
        else:
            ax.annotate(
                "", xy=(r * np.cos(psi), r * np.sin(psi)), xytext=(0, 0),
                arrowprops=dict(arrowstyle="-|>", color=ORANGE,
                                 linewidth=5, mutation_scale=25),
                zorder=4,
            )

        return fig

    @output
    @render.ui
    def r_readout():
        th = theta()
        r, _ = order_parameter(th)
        return ui.HTML(
            f"<div style='font-size:2rem;color:{TEXT};margin-top:0.75rem;'>"
            f"r &asymp; <strong>{r:.2f}</strong></div>"
        )


app = App(app_ui, server)

The metronomes’ motion can be described by a system of them:

\[\frac{d\theta_i}{dt} = \omega_i + \frac{K}{N}\sum_{j=1}^{N} \sin(\theta_j - \theta_i)\]

This is the equation the demo was running – and the little panel top right is that same system running live right now, no controls, fixed below the critical coupling so it keeps drifting rather than settling. Pause here and point at the moving dots: each one is a theta_i(t), and what you’re watching is this equation being numerically integrated in real time. How fast oscillator i’s phase moves – its own natural rhythm omega_i, plus a pull from everyone else, strength K. Frame the equation itself as “here’s the four-line description of what you just watched a few slides ago,” not a thing to be understood, just recognised.

Complex analysis

The order parameter r you just watched is a complex number – its size is “how in sync,” its angle is “the shared rhythm.”

\[r\,e^{i\psi} = \frac{1}{N}\sum_{j=1}^{N} e^{i\theta_j}\]

r is literally the two “Before”/“After” diagrams from a few slides back, written as one complex number: its size r is how clustered the phases are, its angle psi is the shared rhythm they’ve locked onto. The schematic above makes both pieces of the equation visible at once: each theta_j is one dot’s angle on the unit circle, and r e^(i psi) – the orange arrow – is their average as a single complex number, with its own length r and angle psi.

Dynamical systems & bifurcation

Below a critical coupling strength, synchrony never happens.
Above it, it suddenly does.
That sudden flip has a name: a bifurcation.

Order parameter r at steady state, as a function of coupling K, for the same spread-0.7 noisy frequency distribution as the Sync Dial’s default (averaged over several random draws – the shaded band shows how much that steady-state r varies from one random draw to another). Flat and low, then a sudden climb around K_c: that’s the bifurcation. Point back at the Sync Dial slide – this is the curve the room was tracing out live by dragging K.

Network science

Real synchronising systems – power grids, neurons, cardiac tissue, the fireflies – aren’t tidy pairs. They’re networks. Whether they sync at all depends on the properties of the network.
